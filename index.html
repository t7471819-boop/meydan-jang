============================================================ class Battlefield(Entity): """زمین چمنی ناهموار برای میدان جنگ""" def __init__(self, level=1, size=80): super().__init__() self.level = level self.size = size self.hills = [] self.obstacles = [] # جلوتر از بازیکن نباشن (فاصله از مرکز) if abs(x) < 6 and abs(z) < 6: continue sx = random.uniform(3, 7) sy = random.uniform(0.8, 2.5) sz = random.uniform(3, 7) hill = Entity( model='sphere', color=COLOR_GRASS, position=(x, -sy/2 + 0.3, z), scale=(sx, sy, sz), collider='sphere' ) self.hills.append(hill) def _create_obstacles(self): """موانع (مکعب‌های چوبی/سنگی)""" num_obstacles = self.level * 2 for _ in range(num_obstacles): x = random.uniform(-self.size/2 + 5, self.size/2 - 5) z = random.uniform(-self.size/2 + 5, self.size/2 - 5) if abs(x) < 7 and abs(z) < 7: continue obstacle = Entity( model='cube', color=color.rgb(110, 85, 60), position=(x, 1, z), scale=(random.uniform(1, 2.5), 2, random.uniform(1, 2.5)), collider='box', texture='brick' ) self.obstacles.append(obstacle) def cleanup(self): """پاک کردن زمین برای مرحله بعد""" for h in self.hills: destroy(h) for o in self.obstacles: destroy(o) destroy(self.ground) destroy(self.sky) destroy(self.sun) destroy(self) # ============================================================ # بازیکن # ============================================================ class Player(FirstPersonController): def __init__(self, position=(0, 5, 0)): super().__init__( position=position, speed=PLAYER_SPEED, jump_height=PLAYER_JUMP_HEIGHT, gravity=PLAYER_GRAVITY, mouse_sensitivity=Vec2(40, 40), ) self.health = PLAYER_START_HEALTH self.vest = 0 self.money = 0 self.cursor.visible = True self.is_crouching = False self.is_zooming = False self.normal_speed = PLAYER_SPEED self.crouch_speed = PLAYER_SPEED * 0.5 def take_damage(self, damage_to_vest=4, damage_to_health=1): """گرفتن دمیج: اول جلیقه، بعد جون""" if self.vest > 0: # جلیقه دمیج ۲ برابر می‌گیره vest_damage = damage_to_vest if self.vest >= vest_damage: self.vest -= vest_damage print(f'جلیقه: -{vest_damage} → {self.vest}') return else: # جلیقه تموم شد، مازاد به جون می‌ره remaining = vest_damage - self.vest self.vest = 0 self.health -= damage_to_health print(f'جلیقه تمام! جون: {self.health}') else: self.health -= damage_to_health print(f'جون: -{damage_to_health} → {self.health}') # افکت قرمز (فلاش) self._damage_flash() if self.health <= 0: self.health = 0 self.die() def _damage_flash(self): """فلاش قرمز موقع دمیج""" flash = Entity( parent=camera.ui, model='quad', color=color.rgba(255, 0, 0, 0.3), scale=(2, 1), z=-0.9 ) flash.animate_color(color.rgba(255, 0, 0, 0), duration=0.3) destroy(flash, delay=0.35) def add_health(self, amount): self.health = min(self.health + amount, PLAYER_MAX_HEALTH) def add_vest(self, amount): self.vest = min(self.vest + amount, VEST_MAX) def add_money(self, amount): self.money += amount def crouch(self): """نشستن""" if not self.is_crouching: self.is_crouching = True self.speed = self.crouch_speed self.scale_y = 0.6 self.y -= 0.4 def stand_up(self): """بلند شدن""" if self.is_crouching: self.is_crouching = False self.speed = self.normal_speed self.scale_y = 1 self.y += 0.4 def die(self): game_state['game_lost'] = True print('☠️ باختی!') بخش ۳: دشمن‌ها # ============================================================ # ============================================================ # افکت خون (موقع مرگ دشمن) # ============================================================ class BloodEffect(Entity): """افکت پاشیدن خون""" def __init__(self, position): super().__init__( model='sphere', color=color.rgb(180, 0, 0), scale=0.15, position=position ) self.life = 0.5 self.particles = [] # اگه به زمین رسید if p.y < 0.1: p.y = 0.1 p.velocity = Vec3(0, 0, 0) p.color = color.rgba(120, 0, 0, 0.5) if self.life <= 0: for p in self.particles: destroy(p) destroy(self) # ============================================================ # دشمن # ============================================================ class Enemy(Entity): """دشمن با بدن مکعبی و سر دایره‌ای""" def __init__(self, position=(0, 0, 0), armed=False, level=1): super().__init__( position=position, collider='box' ) # ====== مشخصات ====== self.armed = armed self.level = level # سختی بر اساس مرحله self.max_health = 3 + (level // 5) # هر ۵ مرحله +۱ جون self.health = self.max_health self.speed = random.uniform(1.5, 2.5) + (level * 0.05) self.attack_range = 15 self.fire_cooldown = random.uniform(0.5, 1.5) self.fire_rate = max(0.5, 1.5 - (level * 0.02)) self.hit_chance = 0.6 + min(0.3, level * 0.01) # دمیج به جلیقه (بر اساس نوع سلاح دشمن) self.vest_damage = random.choice([2, 4, 4, 6]) # ====== انیمیشن دمیج ====== self.damage_shake_time = 0 self.damage_shake_duration = 0.2 self.shake_intensity = 0.15 self.original_shake_y = 0 # ====== انیمیشن مرگ ====== self.is_dying = False self.death_time = 0 self.death_duration = 2.5 # چقدر طول بکشه تا محو شه # ====== حالت پرتاب (برای افتادن) ====== self.fall_rotation = 0 self.fall_target = -90 # میفته عقب # ====== ساخت بدن ====== self._build_body() # اضافه به لیست دشمن‌ها game_state['enemies'].append(self) def _build_body(self): """ساخت بدن دشمن از مکعب و دایره""" # === بدن (پیراهن سبز نظامی - مکعب) === self.body = Entity( parent=self, model='cube', color=COLOR_UNIFORM, scale=(0.7, 1.0, 0.4), position=(0, 0.5, 0), texture=None ) # === سر (دایره - پوست) === self.head = Entity( parent=self, model='sphere', color=COLOR_SKIN, scale=0.35, position=(0, 1.3, 0) ) # === چشم‌ها (دو دایره‌ی کوچیک مشکی) === self.eye_l = Entity( parent=self, model='sphere', color=color.black, scale=0.06, position=(-0.08, 1.32, 0.16) ) self.eye_r = Entity( parent=self, model='sphere', color=color.black, scale=0.06, position=(0.08, 1.32, 0.16) ) # === کلاه سربازی (مکعب سبز تیره) === self.hat = Entity( parent=self, model='cube', color=COLOR_HAT, scale=(0.42, 0.14, 0.42), position=(0, 1.5, 0) ) # === نقاب کلاه (لبه) === self.hat_brim = Entity( parent=self, model='cube', color=COLOR_HAT, scale=(0.44, 0.04, 0.2), position=(0, 1.44, 0.15) ) # === شلوار (مکعب سبز تیره) === self.pants = Entity( parent=self, model='cube', color=COLOR_PANTS, scale=(0.65, 0.6, 0.4), position=(0, -0.15, 0) ) # === پوتین چپ (مکعب مشکی) === self.boot_l = Entity( parent=self, model='cube', color=COLOR_BOOT, scale=(0.22, 0.18, 0.4), position=(-0.18, -0.55, 0) ) # === پوتین راست (مکعب مشکی) === self.boot_r = Entity( parent=self, model='cube', color=COLOR_BOOT, scale=(0.22, 0.18, 0.4), position=(0.18, -0.55, 0) ) # === کوله پشتی سبز (پشت بدن) === self.backpack = Entity( parent=self, model='cube', color=COLOR_BACKPACK, scale=(0.5, 0.7, 0.25), position=(0, 0.55, -0.32) ) # === دست چپ (دایره - پوست) === self.arm_l = Entity( parent=self, model='sphere', color=COLOR_SKIN, scale=0.18, position=(-0.45, 0.7, 0) ) # === دست راست (دایره - پوست) === self.arm_r = Entity( parent=self, model='sphere', color=COLOR_SKIN, scale=0.18, position=(0.45, 0.7, 0) ) # === اسلحه (اگه مسلح باشه) === if self.armed: self.gun = Entity( parent=self, model='cube', color=color.rgb(30, 30, 30), scale=(0.08, 0.08, 0.55), position=(0.45, 0.7, 0.3) ) # درخشش دهانه (وقتی شلیک می‌کنه) self.muzzle = Entity( parent=self, model='sphere', color=color.rgba(255, 200, 50, 0.9), scale=0.12, position=(0.45, 0.7, 0.6), enabled=False ) # ======================================================== # به‌روزرسانی هر فریم # ======================================================== def update(self): # اگه داره می‌میره، انیمیشن مرگ اجرا شه if self.is_dying: self._update_death_animation() return # اگه بازی تموم شده، کاری نکن if game_state['game_lost'] or game_state['game_won']: return # انیمیشن دمیج if self.damage_shake_time > 0: self.damage_shake_time -= time.dt # لرزش تصادفی self.rotation_z = math.sin(time.time() * 60) * self.shake_intensity * 20 self.y = self.original_shake_y + math.sin(time.time() * 60) * self.shake_intensity * 0.3 else: self.rotation_z = 0 # فاصله تا بازیکن dist = distance(self, player) # حرکت به سمت بازیکن if dist > 2.5: direction = (player.position - self.position).normalized() direction.y = 0 self.position += direction * self.speed * time.dt # چرخش به سمت بازیکن self.look_at(Vec3(player.x, self.y, player.z)) # اگه مسلح بود و در محدوده بود، شلیک کنه if self.armed and dist < self.attack_range: self.fire_cooldown -= time.dt if self.fire_cooldown <= 0: self.fire_cooldown = self.fire_rate self.shoot_at_player() # اگه دست خالی بود و رسید، ضربه بزنه (اختیاری) elif not self.armed and dist < 2: self.fire_cooldown -= time.dt if self.fire_cooldown <= 0: self.fire_cooldown = 1.0 # ضربه‌ی دست خالی - دمیج کمتر player.take_damage(damage_to_vest=2, damage_to_health=1) # ======================================================== # شلیک به بازیکن # ======================================================== def shoot_at_player(self): """شلیک به بازیکن""" # افکت muzzle flash if hasattr(self, 'muzzle'): self.muzzle.enabled = True invoke(setattr, self.muzzle, 'enabled', False, delay=0.08) # احتمال اصابت if random.random() < self.hit_chance: player.take_damage( damage_to_vest=self.vest_damage, damage_to_health=1 ) # ======================================================== # گرفتن دمیج # ======================================================== def take_damage(self, amount=1): """گرفتن دمیج از بازیکن""" if self.is_dying: return self.health -= amount # انیمیشن دمیج (لرزش) self._play_damage_animation() # اگه مرد if self.health <= 0: self.die() def _play_damage_animation(self): """انیمیشن دمیج - تکون + قرمز شدن""" self.damage_shake_time = self.damage_shake_duration self.original_shake_y = self.y # همه‌ی اعضای بدن قرمز بشن original_colors = [] parts = [self.body, self.head, self.pants, self.arm_l, self.arm_r, self.boot_l, self.boot_r] for part in parts: original_colors.append((part, part.color)) part.color = color.rgb(255, 80, 80) # برگردونن به رنگ اصلی def restore(): for part, col in original_colors: if part.enabled: part.color = col invoke(restore, delay=0.15) # ======================================================== # مرگ # ======================================================== def die(self): """شروع انیمیشن مرگ""" if self.is_dying: return self.is_dying = True self.death_time = 0 # افکت خون blood = BloodEffect(position=self.position + Vec3(0, 1, 0)) blood.update = lambda: self._update_blood(blood) def _update_blood(self, blood): """آپدیت افکت خون""" blood.life -= time.dt for p in blood.particles: if not p.enabled: continue p.velocity.y -= 12 * time.dt p.position += p.velocity * time.dt if p.y < 0.1: p.y = 0.1 p.velocity = Vec3(0, 0, 0) if blood.life <= 0: for p in blood.particles: destroy(p) destroy(blood) def _update_death_animation(self): """انیمیشن مرگ: افتادن + محو شدن""" self.death_time += time.dt # مرحله ۱: افتادن به عقب (۰ تا ۰.۸ ثانیه) if self.death_time < 0.8: progress = self.death_time / 0.8 # چرخش نرم به عقب self.rotation_x = -90 * progress # یه ذره هم پایین‌تر self.y = max(0.2, 1 - progress * 0.8) # مرحله ۲: ثابت روی زمین (۰.۸ تا ۱.۵ ثانیه) elif self.death_time < 1.5: self.rotation_x = -90 self.y = 0.2 # مرحله ۳: محو شدن نرم (۱.۵ تا ۲.۵ ثانیه) else: progress = (self.death_time - 1.5) / 1.0 alpha = 1 - progress # کم کردن شفافیت همه‌ی اعضا all_parts = [ self.body, self.head, self.pants, self.arm_l, self.arm_r, self.boot_l, self.boot_r, self.hat, self.hat_brim, self.backpack, self.eye_l, self.eye_r ] if self.armed and hasattr(self, 'gun'): all_parts.append(self.gun) for part in all_parts: if part and part.enabled: c = part.color part.color = color.rgba(c[0], c[1], c[2], alpha) # وقتی کامل محو شد، حذف if progress >= 1.0: self._remove() def _remove(self): """حذف کامل دشمن""" if self in game_state['enemies']: game_state['enemies'].remove(self) # دادن پول به بازیکن reward = 200 + (self.level * 50) player.add_money(reward) destroy(self) # ============================================================ # سیستم ساخت موج دشمن‌ها # ============================================================ class EnemySpawner: """مدیریت ساخت دشمن‌ها""" def __init__(self): self.spawned = 0 self.spawn_radius = 35 def spawn_wave(self, level, count): """ساخت موج دشمن""" for i in range(count): # ۵۰٪ مسلح armed = random.random() < 0.5 # ساخت دشمن enemy = Enemy( position=(x, 1, z), armed=armed, level=level ) self.spawned += 1 print(f'👥 موج دشمن: {count} تا (کل: {self.spawned})') # ============================================================ # شروع موج اول # ============================================================ enemy_spawner = EnemySpawner() # توی مرحله ۱، ۵ دشمن enemy_spawner.spawn_wave( level=game_state['level'], count=5 ) # ============================================================ # ============================================================ # بخش ۴: انفجار و تجهیزات (نارنجک + ماشین زرهی) # ============================================================ # ============================================================ # افکت انفجار # ============================================================ class Explosion(Entity): """انفجار با دود + جرقه + شعاع تخریب""" def __init__(self, position, radius=45, damage=10): super().__init__(position=position) self.radius = radius self.damage = damage self.life = 0 self.duration = 1.5 # ====== کره‌ی آتش ====== self.fireball = Entity( parent=self, model='sphere', color=color.rgba(255, 200, 50, 0.9), scale=0.5, position=(0, 0, 0) ) # ====== کره‌ی نارنجی (لایه‌ی دوم) ====== self.fireball2 = Entity( parent=self, model='sphere', color=color.rgba(255, 100, 0, 0.7), scale=0.8, position=(0, 0, 0) ) #
